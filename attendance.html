from pyscript import document import numpy as np import matplotlib.pyplot as plt import logging logging.getLogger('matplotlib').setLevel(logging.ERROR) days = [] absences = [] def displaying(event): day = document.getElementById("day").value absence = document.getElementById("absence").value if day == "" or absence == "": return absence = int(absence) days.append(day) absences.append(absence) converted_absences = np.array(absences) plt.clf() plt.plot(days, converted_absences, marker='o') plt.title("Weekly Attendance (Absences)") plt.xlabel("Day") plt.ylabel("Number of Absences") plt.grid() display(plt, target="plot", append=False) document.getElementById("day").value = "" document.getElementById("absence").value = ""
